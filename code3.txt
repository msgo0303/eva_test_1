/**
 * 1. config 시트에서 봇 토큰 및 챗 ID 정보를 가져오는 함수
 */
function getBotConfig() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("config");
  const token = sheet.getRange("B1").getValue();
  const chatId = sheet.getRange("B2").getValue();
  return { token: token, chatId: chatId };
}

/**
 * 2. 텔레그램 웹훅 신호 처리 함수 (회원 명부 실시간 수집 및 동적 청소 기믹 결합)
 */
function doPost(e) {
  const lock = LockService.getScriptLock();
  
  try {
    lock.waitLock(10000); 
    
    const config = getBotConfig();
    const responseData = JSON.parse(e.postData.contents);
    
    // 🛡️ [철통 보안 필터 추가] 
    // config 시트 B2(config.chatId)가 비어있거나, 현재 신호가 들어온 방 ID와 다르면 아예 무시합니다.
    let incomingChatId = null;
    if (responseData.message) {
      incomingChatId = responseData.message.chat.id;
    } else if (responseData.callback_query && responseData.callback_query.message) {
      incomingChatId = responseData.callback_query.message.chat.id;
    }

    // 시트 B2의 값과 현재 방 ID를 비교 (숫자/문자열 타입 다를 수 있으니 == 로 비교)
    if (!config.chatId || incomingChatId == null || incomingChatId != config.chatId) {
      // 지정된 우리 단체방이 아니면 봇은 아무 일도 하지 않고 그 자리에서 셔터를 내립니다.
      return; 
    }
    // ----------------------------------------------------

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const attSheet = ss.getSheetByName("attendance");
    const noticeSheet = ss.getSheetByName("notices");
    const userSheet = ss.getSheetByName("Userid");
    const configSheet = ss.getSheetByName("config"); 

    // 하위 코드들 동일... 

    // [방어선 1] 요청이 들어오자마자 1분이 지나도록 방치된 대기열 유저부터 칼같이 청소 (트리거 프리)
    cleanTimeoutRows(config.token, config.chatId, attSheet, noticeSheet);

    // ==========================================
    // Case A: 사용자가 일반 명령어 입력 또는 '답장'을 보냈을 때 (눈팅 및 일반 채팅 포함)
    // ==========================================
    if (responseData.message) {
      const chatId = responseData.message.chat.id;
      const text = responseData.message.text ? responseData.message.text : "";
      const userId = responseData.message.from.id; 
      const username = responseData.message.from.username ? "@" + responseData.message.from.username : ""; 

      // 성과 이름을 하나로 찰떡 결합
      const uFirstName = responseData.message.from.first_name || "";
      const uLastName = responseData.message.from.last_name || "";
      const fullName = (uLastName + uFirstName).trim() || "이름없음";

      // [명부 수집] 일반 채팅 감지 시 회원 명부(Userid 시트)에 자동 기록/최신화
      updateUserDatabase(userSheet, userId, fullName, username);

      // 1. 출석부 생성 명령어 처리 (/create [공지내용])
      if (text.startsWith("/create")) {
        let notice = "";
        if (text.length > 7) {
          notice = text.substring(7).trim(); 
        }
        sendAttendanceKeyboard(config.token, chatId, notice, noticeSheet, configSheet);
        return;
      }

      // 2. 공지사항 실시간 수정 명령어 (/update [새로운공지내용])
      else if (text.startsWith("/update")) {
        let newNotice = "";
        if (text.length > 7) {
          newNotice = text.substring(7).trim();
        }

        let targetMessageId = null;
        if (responseData.message.reply_to_message) {
          targetMessageId = responseData.message.reply_to_message.message_id; 
        } else {
          const noticeData = noticeSheet.getDataRange().getValues();
          if (noticeData.length > 1) {
            targetMessageId = noticeData[noticeData.length - 1][0]; 
          }
        }

        if (targetMessageId) {
          updateNoticeDb(noticeSheet, targetMessageId, newNotice);
          updateAttendanceMessage(config.token, chatId, targetMessageId, attSheet, noticeSheet);
          deleteMessage(config.token, chatId, responseData.message.message_id);
        }
        return;
      }

      // 3. 최신 출석부 위치로 워프하는 명령어 (/go)
      else if (text.startsWith("/go")) {
        const noticeData = noticeSheet.getDataRange().getValues();
        if (noticeData.length > 1) {
          const lastMessageId = noticeData[noticeData.length - 1][0];
          
          const url = "https://api.telegram.org/bot" + config.token + "/sendMessage";
          const payload = {
            "chat_id": chatId,
            "text": "📢 최신 출석부 투표는 바로 위 메시지(답장 링크)를 터치해서 진행해 주세요!",
            "reply_to_message_id": lastMessageId 
          };
          
          const options = { "method": "post", "contentType": "application/json", "payload": JSON.stringify(payload) };
          UrlFetchApp.fetch(url, options);
        } else {
          sendTelegramMessage(config.token, chatId, "❌ 현재 방에 생성된 출석부가 없습니다. 먼저 /create 명령어로 출석부를 만들어 주세요.");
        }
        return;
      }

      // 4. 사유 입력 요구에 '답장'을 한 경우 (불참/부분참석 사유 최종 확정)
      if (responseData.message.reply_to_message) {
        const replyText = responseData.message.reply_to_message.text;
        
        if (replyText.includes("사유를 입력해주세요")) {
          const data = attSheet.getDataRange().getValues();
          let userRowIndex = -1;
          let targetMessageId = null;
          let currentStatus = "";

          for (let i = data.length - 1; i >= 1; i--) {
            if (data[i][1] == userId) {
              userRowIndex = i + 1;
              targetMessageId = data[i][0];
              currentStatus = data[i][3];
              break;
            }
          }

          if (userRowIndex !== -1 && targetMessageId) {
            let finalStatus = currentStatus;
            if (currentStatus === "부분_대기") finalStatus = "부분"; 
            if (currentStatus === "불참_대기") finalStatus = "불참";

            attSheet.getRange(userRowIndex, 3).setValue(fullName);    
            attSheet.getRange(userRowIndex, 4).setValue(finalStatus); 
            attSheet.getRange(userRowIndex, 5).setValue(text);        
            
            updateAttendanceMessage(config.token, chatId, targetMessageId, attSheet, noticeSheet);
            
            deleteMessage(config.token, chatId, responseData.message.reply_to_message.message_id);
            deleteMessage(config.token, chatId, responseData.message.message_id);
          }
        }
      }
    }
    
    // ==========================================
    // Case B: 사용자가 '인라인 버튼'을 클릭했을 때
    // ==========================================
    else if (responseData.callback_query) {
      const callbackQuery = responseData.callback_query;
      const callbackQueryId = callbackQuery.id;
      const messageId = callbackQuery.message.message_id; 
      const chatId = callbackQuery.message.chat.id;
      const userId = callbackQuery.from.id;               

      const cbFirstName = callbackQuery.from.first_name || "";
      const cbLastName = callbackQuery.from.last_name || "";
      const fullName = (cbLastName + cbFirstName).trim() || "이름없음";
      const username = callbackQuery.from.username ? "@" + callbackQuery.from.username : "";
      let action = callbackQuery.data;

      // [명부 수집] 버튼 클릭 시점에도 회원 명부 자동 기록/최신화
      updateUserDatabase(userSheet, userId, fullName, username);

      const data = attSheet.getDataRange().getValues();
      let userRowIndex = -1;
      let existingStatus = "";
      let existingDetail = "";

      for (let i = 1; i < data.length; i++) {
        if (data[i][0] == messageId && data[i][1] == userId) {
          userRowIndex = i + 1; 
          existingStatus = data[i][3];
          existingDetail = data[i][4]; 
          break;
        }
      }

      const now = new Date();
      const hours = String(now.getHours()).padStart(2, '0');
      const minutes = String(now.getMinutes()).padStart(2, '0');
      const timeStr = hours + ":" + minutes;

      if ((action === "부분" && existingStatus === "부분_대기") || 
          (action === "불참" && existingStatus === "불참_대기")) {
        answerCallbackQuery(config.token, callbackQueryId, "이미 사유 입력 안내 메시지가 발송되었습니다.");
        return;
      }

      if (existingStatus.endsWith("_대기") && existingDetail && !isNaN(existingDetail)) {
        deleteMessage(config.token, chatId, existingDetail); 
      }

      if (userRowIndex !== -1) {
        attSheet.getRange(userRowIndex, 3).setValue(fullName);
        
        if (action === "도착") {
          if (existingStatus !== "부분") {
            attSheet.getRange(userRowIndex, 5).setValue(""); 
          }
          attSheet.getRange(userRowIndex, 4).setValue(action);    
          attSheet.getRange(userRowIndex, 6).setValue(now);       
          attSheet.getRange(userRowIndex, 7).setValue(timeStr); 
        } else if (action === "참석") {
          attSheet.getRange(userRowIndex, 4).setValue(action);    
          attSheet.getRange(userRowIndex, 5).setValue(""); 
          attSheet.getRange(userRowIndex, 6).setValue(now);       
          attSheet.getRange(userRowIndex, 7).setValue("");        
        } else {
          let dbStatus = action === "부분" ? "부분_대기" : "불참_대기"; 
          attSheet.getRange(userRowIndex, 4).setValue(dbStatus);  
          attSheet.getRange(userRowIndex, 5).setValue(""); 
          attSheet.getRange(userRowIndex, 6).setValue(now);       
          attSheet.getRange(userRowIndex, 7).setValue("");        
        }
      } else {
        if (action === "도착") {
          attSheet.appendRow([messageId, userId, fullName, action, "", now, timeStr]);
        } else if (action === "참석") {
          attSheet.appendRow([messageId, userId, fullName, action, "", now, ""]);
        } else {
          let dbStatus = action === "부분" ? "부분_대기" : "불참_대기";
          attSheet.appendRow([messageId, userId, fullName, dbStatus, "", now, ""]);
        }
      }

      answerCallbackQuery(config.token, callbackQueryId, fullName + "님 요청 처리 중...");

      if (action === "참석" || action === "도착") {
        updateAttendanceMessage(config.token, chatId, messageId, attSheet, noticeSheet);
      } 
      else if (action === "부분") {
        const mentionText = "⏱️ [" + fullName + "](tg://user?id=" + userId + ")님 부분참석 시간과 사유를 입력해주세요.\n(예시: 16:00-17:00/직장)\n이 메시지에 '답장' 기능으로 입력해주세요!";
        const sentMsg = sendForceReply(config.token, chatId, mentionText);
        if (sentMsg) {
          if (userRowIndex === -1) userRowIndex = attSheet.getLastRow();
          attSheet.getRange(userRowIndex, 5).setValue(sentMsg.message_id); 
        }
      } 
      else if (action === "불참") {
        const mentionText = "❌ [" + fullName + "](tg://user?id=" + userId + ")님 불참 사유를 입력해주세요.\n(예시: 개인사정)\n이 메시지에 '답장' 기능으로 입력해주세요!";
        const sentMsg = sendForceReply(config.token, chatId, mentionText);
        if (sentMsg) {
          if (userRowIndex === -1) userRowIndex = attSheet.getLastRow();
          attSheet.getRange(userRowIndex, 5).setValue(sentMsg.message_id); 
        }
      }
    }

  } catch (error) {
    Logger.log("doPost Error: " + error.toString());
  } finally {
    lock.releaseLock(); 
  }
}

/**
 * 📊 회원 명부(Userid 시트) 실시간 관리 함수
 */
function updateUserDatabase(userSheet, userId, fullName, username) {
  const userData = userSheet.getDataRange().getValues();
  const now = new Date();
  let existRowIndex = -1;

  for (let i = 1; i < userData.length; i++) {
    if (userData[i][0] == userId) {
      existRowIndex = i + 1;
      break;
    }
  }

  if (existRowIndex !== -1) {
    userSheet.getRange(existRowIndex, 2).setValue(fullName);   
    userSheet.getRange(existRowIndex, 3).setValue(username);   
    userSheet.getRange(existRowIndex, 4).setValue(now); 
  } else {
    userSheet.appendRow([userId, fullName, username, now, now]);
  }
}

/**
 * 🛡️ 구글 트리거 없이 동작하는 '실시간 동적 청소기' 함수
 */
function cleanTimeoutRows(token, chatId, attSheet, noticeSheet) {
  const data = attSheet.getDataRange().getValues();
  const now = new Date();
  let needToRefreshMessageIds = {}; 

  for (let i = data.length - 1; i >= 1; i--) {
    const status = data[i][3]; 
    const detailMessageId = data[i][4]; 
    const timestamp = new Date(data[i][5]); 

    if (status && status.endsWith("_대기") && !isNaN(detailMessageId) && detailMessageId !== "") {
      const diffMs = now.getTime() - timestamp.getTime();
      
      if (diffMs >= 60000) {
        const targetMessageId = data[i][0]; 
        
        deleteMessage(token, chatId, detailMessageId); 
        attSheet.deleteRow(i + 1); 
        
        needToRefreshMessageIds[targetMessageId] = true; 
      }
    }
  }

  for (let targetMessageId in needToRefreshMessageIds) {
    updateAttendanceMessage(token, chatId, targetMessageId, attSheet, noticeSheet);
  }
}

/**
 * 3. [기믹 튜닝 완료] 출석부 메시지 새로고침 함수 (이미 투표한 유저를 취합전 명단에서 동적 제외)
 */
function updateAttendanceMessage(token, chatId, messageId, sheet, noticeSheet) {
  const data = sheet.getDataRange().getValues();
  
  let arrived = [];
  let attending = [];
  let partial = [];
  let absent = [];
  
  // 🔥 이 출석부에 이미 투표를 완료한 유저들의 ID를 가로챌 세트(Set) 객체 생성
  let votedUserIds = {};

  for (let i = 1; i < data.length; i++) {
    if (data[i][0] == messageId) {
      const uId = String(data[i][1]); // B열: User_ID
      const name = data[i][2]; 
      const status = data[i][3];
      let detail = data[i][4];
      let arriveTime = data[i][6]; 

      // 투표를 아예 안 한 가상 대기열 상태가 아니라면, 투표자 명단 풀에 ID 등록
      if (status && !status.endsWith("_대기")) {
        votedUserIds[uId] = true;
      }

      if (detail instanceof Date) {
        const hours = String(detail.getHours()).padStart(2, '0');
        const minutes = String(detail.getMinutes()).padStart(2, '0');
        detail = hours + ":" + minutes;
      } else if (detail) {
        detail = String(detail);
      }

      if (status.endsWith("_대기") && !isNaN(detail)) {
        detail = "";
      }

      if (arriveTime instanceof Date) {
        const hours = String(arriveTime.getHours()).padStart(2, '0');
        const minutes = String(arriveTime.getMinutes()).padStart(2, '0');
        arriveTime = hours + ":" + minutes;
      } else if (arriveTime) {
        arriveTime = String(arriveTime);
      }

      if (status === "도착") {
        if (detail) {
          arrived.push(name + "(" + arriveTime + " 도착 | 부분참 " + detail + ")");
        } else {
          arrived.push(name + "(" + arriveTime + ")");
        }
      } else if (status === "참석") {
        attending.push(name);
      } else if (status.startsWith("부분")) { 
        partial.push({ name: name, detail: detail });
      } else if (status.startsWith("불참")) { 
        absent.push(name + (detail ? "/" + detail : ""));
      }
    }
  }

  partial.sort(function(a, b) {
    return a.detail.localeCompare(b.detail);
  });

  let partialFormatted = partial.map(function(item) {
    return item.name + (item.detail ? "(" + item.detail + ")" : "");
  });

  // 👥 [핵심 고도화] config 명단판을 다시 열어, 이미 투표한 사람(votedUserIds)을 실시간으로 도려냅니다.
  const configSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("config");
  const configData = configSheet.getDataRange().getValues();
  let tagList = [];

  for (let i = 1; i < configData.length; i++) {
    const row = configData[i];
    if (row.length >= 10) {
      let name = row[7];       // H열: 이름
      const userId = String(row[8]); // I열: ID
      const isTarget = row[9]; // J열: 태그대상

      if (isTarget === true || String(isTarget).toUpperCase() === "TRUE") {
        if (name && userId) {
          // 🛡️ 마크다운 꼬임 방어선 적용
          name = String(name).replace(/[\[\]\(\)\_\*]/g, "").trim();
          
          // 💡 만약 이 유저 ID가 '이미 투표한 사람 목록'에 없다면 취합 전 명단에 유지시킵니다!
          if (!votedUserIds[userId]) {
            tagList.push("[" + name + "](tg://user?id=" + userId + ")");
          }
        }
      }
    }
  }

  // 원본 순수 공지사항만 긁어옵니다 (기존에 쳐놨던 텍스트 슬라이싱 방어)
  let rawNotice = getNoticeFromDb(noticeSheet, messageId);
  if (rawNotice.includes("---취합 전---")) {
    rawNotice = rawNotice.split("---취합 전---")[0].trim();
  }

  // 아직 투표 안 한 남은 미지출자 명단이 있을 때만 하단에 꼬리표 새로 조립
  let fullNoticeBody = rawNotice;
  if (tagList.length > 0) {
    fullNoticeBody += "\n\n---취합 전---\n" + tagList.join(", ");
  }

  const today = new Date();
  const dateStr = String(today.getMonth() + 1).padStart(2, '0') + "-" + String(today.getDate()).padStart(2, '0');

  let text = dateStr + " 오프찾 활동 현황\n반영이 느려도 조금만 기다려주세요🥹\n";
  if (fullNoticeBody) {
    text += fullNoticeBody + "\n"; 
  }
  text += "\n1. 현장 도착 (" + arrived.length + "명)\n" + (arrived.length > 0 ? arrived.join("\n") : "-") + "\n\n";
  text += "2. 참석 예정 (" + attending.length + "명)\n" + (attending.length > 0 ? attending.join("\n") : "-") + "\n\n";
  text += "3. 부분 참석 예정 (" + partialFormatted.length + "명)\n" + (partialFormatted.length > 0 ? partialFormatted.join("\n") : "-") + "\n\n";
  text += "4. 불참자 (" + absent.length + "명)\n" + (absent.length > 0 ? absent.join("\n") : "-");

  const url = "https://api.telegram.org/bot" + token + "/editMessageText";
  const payload = {
    "chat_id": chatId,
    "message_id": messageId,
    "text": text,
    "parse_mode": "Markdown",
    "reply_markup": {
      "inline_keyboard": [
        [
          { "text": "👍 참석", "callback_data": "참석" },
          { "text": "⏱️ 부분참석", "callback_data": "부분" }
        ],
        [
          { "text": "❌ 불참", "callback_data": "불참" },
          { "text": "🏃 도착", "callback_data": "도착" }
        ]
      ]
    }
  };
  
  const options = { "method": "post", "contentType": "application/json", "payload": JSON.stringify(payload) };
  UrlFetchApp.fetch(url, options);
}

/**
 * 4. 출석부 최초 생성 함수
 */
function sendAttendanceKeyboard(token, chatId, notice, noticeSheet, configSheet) {
  const url = "https://api.telegram.org/bot" + token + "/sendMessage";
  const today = new Date();
  const dateStr = String(today.getMonth() + 1).padStart(2, '0') + "-" + String(today.getDate()).padStart(2, '0');
  
  const configData = configSheet.getDataRange().getValues();
  let tagList = [];

  for (let i = 1; i < configData.length; i++) {
    const row = configData[i];
    if (row.length >= 10) {
      let name = row[7];       
      const userId = row[8];     
      const isTarget = row[9];   

      if (isTarget === true || String(isTarget).toUpperCase() === "TRUE") {
        if (name && userId) {
          name = String(name).replace(/[\[\]\(\)\_\*]/g, "").trim();
          tagList.push("[" + name + "](tg://user?id=" + userId + ")");
        }
      }
    }
  }

  let fullNoticeBody = notice;
  if (tagList.length > 0) {
    fullNoticeBody += "\n\n---취합 전---\n" + tagList.join(", ");
  }
  
  let text = dateStr + " 오프찾 활동 현황\n반영이 느려도 조금만 기다려주세요🥹\n";
  if (fullNoticeBody) {
    text += fullNoticeBody + "\n"; 
  }
  text += "\n1. 현장 도착 (0명)\n-\n\n2. 참석 예정 (0명)\n-\n\n3. 부분 참석 예정 (0명)\n-\n\n4. 불참자 (0명)\n-";
  
  const payload = {
    "chat_id": chatId,
    "text": text,
    "parse_mode": "Markdown", 
    "reply_markup": {
      "inline_keyboard": [
        [
          { "text": "👍 참석", "callback_data": "참석" },
          { "text": "⏱️ 부분참석", "callback_data": "부분" }
        ],
        [
          { "text": "❌ 불참", "callback_data": "불참" },
          { "text": "🏃 도착", "callback_data": "도착" }
        ]
      ]
    }
  };
  const options = { "method": "post", "contentType": "application/json", "payload": JSON.stringify(payload) };
  const response = UrlFetchApp.fetch(url, options);
  
  const resData = JSON.parse(response.getContentText());
  if (resData.ok) {
    const messageId = resData.result.message_id;
    noticeSheet.appendRow([messageId, fullNoticeBody, new Date()]); 
  }
}

/**
 * 5. notices 시트에서 특정 Message_ID에 해당하는 공지를 찾는 헬퍼 함수
 */
function getNoticeFromDb(noticeSheet, messageId) {
  const data = noticeSheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] == messageId) {
      return data[i][1]; 
    }
  }
  return "";
}

/**
 * 6. notices 시트의 특정 Message_ID에 해당하는 공지를 수정 및 덮어쓰기 하는 헬퍼 함수
 */
function updateNoticeDb(noticeSheet, messageId, newNotice) {
  const data = noticeSheet.getDataRange().getValues();
  let foundRow = -1;
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] == messageId) {
      foundRow = i + 1;
      break;
    }
  }
  if (foundRow !== -1) {
    noticeSheet.getRange(foundRow, 2).setValue(newNotice); 
  } else {
    noticeSheet.appendRow([messageId, newNotice, new Date()]);
  }
}

/**
 * 7. 텍스트 입력을 강제하고 생성된 메시지 객체를 반환하는 ForceReply 함수
 */
function sendForceReply(token, chatId, text) {
  const url = "https://api.telegram.org/bot" + token + "/sendMessage";
  const payload = { 
    "chat_id": chatId, 
    "text": text, 
    "parse_mode": "Markdown",
    "reply_markup": { "force_reply": true, "selective": true } 
  };
  const options = { "method": "post", "contentType": "application/json", "payload": JSON.stringify(payload) };
  const res = UrlFetchApp.fetch(url, options);
  return JSON.parse(res.getContentText()).result; 
}

function deleteMessage(token, chatId, messageId) {
  try {
    const url = "https://api.telegram.org/bot" + token + "/deleteMessage";
    const payload = { "chat_id": chatId, "message_id": messageId };
    const options = { "method": "post", "contentType": "application/json", "muteHttpExceptions": true, "payload": JSON.stringify(payload) };
    UrlFetchApp.fetch(url, options);

}
